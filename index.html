חיבור ל-Google Sheets ב-3 צעדים פשוטים

מאפשר סנכרון בזמן אמת של כל השיבוצים בין כל תושבי היישוב בחינם

  1. פתחו גיליון חדש ב-Google Sheets, ובחרו בתפריט העליון: הרחבות (Extensions) ← Apps Script.
  2. מחקו את הקוד הקיים, הדביקו את הקוד שלמטה ולחצו על Deploy ← New deployment ← Web app (הגדירו גישה ל-Anyone).
  3. העתיקו את כתובת ה-Web App שקיבלתם והדביקו אותה בתוך המשתנה GOOGLE_SCRIPT_URL בראש הסקריפט של דף זה.
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var today = new Date().toISOString().slice(0, 10);
  var bookings = [];
  
  for (var i = 1; i < data.length; i++) {
    if (data[i][0] === today) {
      bookings.push({
        id: data[i][1],
        slot: data[i][2],
        dogName: data[i][3],
        ownerName: data[i][4],
        temperament: data[i][5],
        incompatibleWith: data[i][6] ? data[i][6].toString().split(',').map(function(s){ return s.trim(); }).filter(Boolean) : []
      });
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ status: 'success', bookings: bookings }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['Date', 'ID', 'Slot', 'DogName', 'OwnerName', 'Temperament', 'IncompatibleWith']);
  }
  var payload = JSON.parse(e.postData.contents);
  var today = new Date().toISOString().slice(0, 10);
  
  sheet.appendRow([
    today,
    payload.id,
    payload.slot,
    payload.dogName,
    payload.ownerName,
    payload.temperament,
    (payload.incompatibleWith || []).join(', ')
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({ status: 'success' }))
    .setMimeType(ContentService.MimeType.JSON);
}
