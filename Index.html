הצעת עזרה בשמירה / טיול

חיבור ל-Google Sheets ב-3 צעדים פשוטים

מאפשר סנכרון בזמן אמת של כל השיבוצים בין כל תושבי היישוב בחינם

  1. פתחו גיליון חדש ב-Google Sheets, ובחרו בתפריט העליון: הרחבות (Extensions) ← Apps Script.
  2. מחקו את הקוד הקיים, הדביקו את הקוד שלמטה ולחצו על Deploy ← New deployment ← Web app (הגדירו גישה ל-Anyone).
  3. העתיקו את כתובת ה-Web App שקיבלתם והדביקו אותה בתוך המשתנה GOOGLE_SCRIPT_URL בראש הסקריפט של דף זה.
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var today = Utilities.formatDate(new Date(), "Asia/Jerusalem", "yyyy-MM-dd");
  var bookings = [];
  
  for (var i = 1; i < data.length; i++) {
    var rowDate = data[i][0] instanceof Date 
      ? Utilities.formatDate(data[i][0], "Asia/Jerusalem", "yyyy-MM-dd") 
      : String(data[i][0]).trim();
    if (!rowDate || rowDate === today) {
      var slotVal = String(data[i][2] || '').trim();
      var dogVal = String(data[i][3] || '').trim();
      if (slotVal && dogVal) {
        bookings.push({
          id: String(data[i][1] || ('row-' + i)),
          slot: slotVal,
          dogName: dogVal,
          ownerName: String(data[i][4] || '').trim(),
          temperament: String(data[i][5] || 'ידידותי לכולם 🐾').trim(),
          incompatibleWith: data[i][6] ? String(data[i][6]).split(',').map(function(s){ return s.trim(); }).filter(Boolean) : []
        });
      }
    }
  }

  // טעינת מאגר דוג-וולקרים ובקשות שמירה מלשוניות ייעודיות
  var walkers = getSheetRowsAsObjects_(ss, 'DogWalkers', ['id', 'name', 'phone', 'service', 'availability']);
  var careRequests = getSheetRowsAsObjects_(ss, 'CareRequests', ['id', 'phone', 'startDate', 'endDate', 'note', 'offers']);

  var resultJson = JSON.stringify({
    status: 'success',
    bookings: bookings,
    walkers: walkers,
    careRequests: careRequests
  });

  // תמיכה ב-JSONP למניעת חסימות CORS בדפדפן
  if (e && e.parameter && e.parameter.callback) {
    return ContentService.createTextOutput(e.parameter.callback + '(' + resultJson + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(resultJson)
    .setMimeType(ContentService.MimeType.JSON);
}

function getSheetRowsAsObjects_(ss, sheetName, keys) {
  var sh = ss.getSheetByName(sheetName);
  if (!sh) return [];
  var rows = sh.getDataRange().getValues();
  var out = [];
  for (var i = 1; i < rows.length; i++) {
    var obj = {};
    for (var k = 0; k < keys.length; k++) {
      var val = rows[i][k];
      if (val instanceof Date) {
        val = Utilities.formatDate(val, "Asia/Jerusalem", "yyyy-MM-dd");
      }
      obj[keys[k]] = String(val || '').trim();
    }
    if (obj.id) out.push(obj);
  }
  return out;
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var payload = JSON.parse(e.postData.contents);
  var today = Utilities.formatDate(new Date(), "Asia/Jerusalem", "yyyy-MM-dd");

  if (payload.type === 'walker') {
    var wSheet = ss.getSheetByName('DogWalkers') || ss.insertSheet('DogWalkers');
    if (wSheet.getLastRow() === 0) {
      wSheet.appendRow(['ID', 'Name', 'Phone', 'Service', 'Availability', 'CreatedAt']);
    }
    wSheet.appendRow([payload.id, payload.name, payload.phone, payload.service, payload.availability, today]);
  } else if (payload.type === 'careRequest') {
    var cSheet = ss.getSheetByName('CareRequests') || ss.insertSheet('CareRequests');
    if (cSheet.getLastRow() === 0) {
      cSheet.appendRow(['ID', 'Phone', 'StartDate', 'EndDate', 'Note', 'Offers', 'CreatedAt']);
    }
    cSheet.appendRow([payload.id, payload.phone, payload.startDate, payload.endDate, payload.note || '', payload.offers || '', today]);
  } else if (payload.type === 'offerHelp') {
    var cSheet2 = ss.getSheetByName('CareRequests');
    if (cSheet2) {
      var data = cSheet2.getDataRange().getValues();
      for (var i = 1; i < data.length; i++) {
        if (String(data[i][0]).trim() === String(payload.requestId).trim()) {
          var existingOffers = String(data[i][5] || '').trim();
          var updated = existingOffers ? (existingOffers + ' | ' + payload.offerText) : payload.offerText;
          cSheet2.getRange(i + 1, 6).setValue(updated);
          break;
        }
      }
    }
  } else {
    var sheet = ss.getActiveSheet();
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(['Date', 'ID', 'Slot', 'DogName', 'OwnerName', 'Temperament', 'IncompatibleWith']);
    }
    sheet.appendRow([
      today,
      payload.id,
      payload.slot,
      payload.dogName,
      payload.ownerName,
      payload.temperament,
      (payload.incompatibleWith || []).join(', ')
    ]);
  }
  
  return ContentService.createTextOutput(JSON.stringify({ status: 'success' }))
    .setMimeType(ContentService.MimeType.JSON);
}
